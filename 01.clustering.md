library(ArchR)
library(tidyverse)
library(Seurat)

## anno
library("BSgenome.Mfascicularis.NCBI.5.0")
addArchRThreads(threads = 50)
gene=read.table("/cluster/share/atac_group/huangym/macaca_total/ref/MJ_creat_Gene_Macaca_fascicularis_5.0.91.2.2.V20230616.txt",header = F,stringsAsFactors = FALSE,sep = "\t")
colnames(gene)=c("chr","start","end","strand","gene_id","symbol")
gene$symbol=make.unique(gene$symbol)
gene1=gene[-grep("MT",gene$chr),]
MF_gene=makeGRangesFromDataFrame(gene1,keep.extra.columns = T)

exon=read.table("/cluster/share/atac_group/huangym/macaca_total/ref/MJ_creat_Exon_Macaca_fascicularis_5.0.91.2.2.V20230616.txt",header = F,stringsAsFactors = FALSE,sep = "\t")
colnames(exon)=c("chr","start","end","strand","gene_id","symbol")
exon$symbol=make.unique(exon$symbol)
exon1=exon[-grep("MT",exon$chr),]
MF_exon=makeGRangesFromDataFrame(exon1,keep.extra.columns = T)

tss=read.table("/cluster/share/atac_group/huangym/macaca_total/ref/MJ_creat_Transcript_Macaca_fascicularis_5.0.91.2.2.V20230616.txt",header = F,stringsAsFactors = FALSE,sep = "\t")
colnames(tss)=c("chr","start","end","strand","tx_id","tx_name")
tss1=tss[-grep("MT",tss$chr),]
MF_tss=makeGRangesFromDataFrame(tss1,keep.extra.columns = T)
MF_tss=resize(MF_tss, 1, fix = "start")

seqnames(BSgenome.Mfascicularis.NCBI.5.0)=gsub("MFA","chr",seqnames(BSgenome.Mfascicularis.NCBI.5.0))
genomeAnnotation <- createGenomeAnnotation(genome = BSgenome.Mfascicularis.NCBI.5.0)
black=as.data.frame(genomeAnnotation$chromSizes)[,-4]
black$start=black$end+1
black$end=black$end+10
black$strand="+"
genomeAnnotation$blacklist=makeGRangesFromDataFrame(black,keep.extra.columns = T)

geneAnnotation <- createGeneAnnotation(
  TSS = MF_tss,
  exons = MF_exon,
  genes = MF_gene
)

#### Sample call peak
data = readRDS("ArchR_project/Save.rds")

data <- addIterativeLSI(
  ArchRProj = data,
  useMatrix = "TileMatrix",
  name = "IterativeLSI",
  iterations = 2,
  clusterParams = list( #See Seurat::FindClusters
    resolution = c(0.8),
    sampleCells = 20000,
    n.start = 10
  ),
  varFeatures = 25000,
  dimsToUse = 1:50,
  force = T
)

data <- addHarmony(
  ArchRProj = data,
  reducedDims = "IterativeLSI",
  name = "Harmony",
  groupBy = "Sample",
  force = TRUE
)

data <- addClusters(
  input = data,
  reducedDims = "IterativeLSI",
  method = "Seurat",
  name = "Clusters",
  resolution = 0.8,
  maxClusters = 50,
  force = TRUE
)

data <- addUMAP(
  ArchRProj = data,
  reducedDims = "Harmony",
  name = "UMAPHarmony",
  nNeighbors = 30,
  minDist = 0.5,
  metric = "cosine",
  force = TRUE
)

#### 
projHeme1 <- addGroupCoverages(ArchRProj = data, groupBy = "Sample",force = T)

pathToMacs2 <- findMacs2()
projHeme1 <- addReproduciblePeakSet(
    ArchRProj = projHeme1, 
    groupBy = "CellType", 
    peaksPerCell = 2000,
    maxPeaks = 500000,
    pathToMacs2 = pathToMacs2,
    genomeSize = 2.9e9,
    force = T
)

addArchRThreads(threads = 20)
projHeme2 = addPeakMatrix(projHeme1)
saveArchRProject(ArchRProj = projHeme2, outputDirectory = "Save_macaque", load = T)
